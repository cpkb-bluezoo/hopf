Access control

CIDR allow / deny

On TcpListenerConfig:

use hopf_core::{IpNet, PeerAcl, TcpListenerConfig};

let acl = PeerAcl {
    allow: vec![], // empty allow = allow all (unless denied)
    deny: vec![IpNet::parse("10.0.0.0/8").unwrap()],
};
config.with_acl(acl);

Deny wins. Non-empty allow means the peer must match at least one network. Rejection closes the accepted socket before ProtocolHandler::connected.

Accept rate limits

use std::time::Duration;
use hopf_core::AcceptRateLimit;

config.with_rate_limit(AcceptRateLimit::new(
    /* per_source */ 100,
    Duration::from_secs(1),
    /* global */ 0, // 0 = unlimited
));
